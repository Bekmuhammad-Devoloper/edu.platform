get_total_value nomli function yozing.

  • Kirish: car_id INT - mashina ID si
  • Chiqish: INT - shu mashina uchun umumiy qiymat
  • Hisoblash: price * count
  • Agar mashina topilmasa, 0 qaytsin
Misol:
SELECT get_total_value(1); → 500000 (BMW: 10000 × 50)
✅ Yechim:
CREATE OR REPLACE FUNCTION get_total_value(p_car_id INT)
   RETURNS INT
   LANGUAGE plpgsql
AS
$$
DECLARE
   total INT;
BEGIN
   SELECT price * count INTO total
   FROM hw4.cars
   WHERE id = p_car_id;
   
   IF total IS NULL THEN
      RETURN 0;
   END IF;
   
   RETURN total;
END;
$$;

-- Ishlatish:
SELECT get_total_value(1);  -- BMW: 10000 * 50 = 500000
SELECT get_total_value(3);  -- TOYOTA: 8000 * 1000 = 8000000
SELECT get_total_value(999); -- 0 (mavjud emas)

get_most_expensive_car nomli function yozing.

  • Kirish: parametr yo'q

CREATE OR REPLACE FUNCTION get_most_expensive_car()
   RETURNS VARCHAR(255)
   LANGUAGE plpgsql
AS
$$
DECLARE
   car_name VARCHAR(255);
BEGIN
   SELECT model INTO car_name
   FROM hw4.cars
   ORDER BY price DESC
   LIMIT 1;
   
   RETURN car_name;
END;
$$;

-- Ishlatish:

📝 Masala: Sotib olish imkoniyati

▼
📋 TOPSHIRIQ:

can_buy_count nomli function yozing.

  • Kirish: user_id INT, car_id INT
  • Chiqish: INT - user nechta mashina sotib olishi mumkin
  • User balansi va mashina narxiga qarang
  • Zaxiradagi sondan ko'p bo'lmasin
  • User yoki mashina topilmasa, 0 qaytsin
Misol:
SELECT can_buy_count(1, 1); → 10
user1 balansi: 100000, BMW narxi: 10000, zaxira: 50
100000 ÷ 10000 = 10 ta (zaxirada 50 ta bor, 10 qaytadi)
✅ Yechim:
CREATE OR REPLACE FUNCTION can_buy_count(p_user_id INT, p_car_id INT)
   RETURNS INT
   LANGUAGE plpgsql
AS
$$
DECLARE
   user_bal INT;
   car_price INT;
   car_available INT;
   can_afford INT;
BEGIN
   -- User balansini olish
   SELECT balance INTO user_bal FROM hw4.users WHERE id = p_user_id;
   
   -- Mashina ma'lumotlarini olish
   SELECT price, count INTO car_price, car_available 
   FROM hw4.cars WHERE id = p_car_id;
   
   -- Tekshiruvlar
   IF user_bal IS NULL OR car_price IS NULL THEN
      RETURN 0;
   END IF;
   
   -- Qanchaga puli yetadi
   can_afford := user_bal / car_price;
   
   -- Minimal qiymatni qaytarish (pul yoki zaxira)
   IF can_afford < car_available THEN
      RETURN can_afford;
   ELSE
      RETURN car_available;
   END IF;
END;
$$;

-- Ishlatish:
SELECT can_buy_count(1, 1);  -- 10
SELECT can_buy_count(4, 1);  -- 1 (user4 balansi: 10000)

   RETURNS TABLE(
      user_id INT,
      user_name VARCHAR(255),
      user_balance INT
   )
   LANGUAGE plpgsql
AS
$$
BEGIN

   FROM hw4.users

   ORDER BY balance DESC;

      RAISE NOTICE '❌ User topilmadi: %', p_user_id;
      ROLLBACK;
      RETURN;
   END IF;

  • Kirish: from_user INT, to_user INT, amount INT

      RETURN;
   END IF;

   new_price := current_price + (current_price * p_percent / 100);

      RETURN;

📝 Masala: Yangi mashina qo'shish

▼
📋 TOPSHIRIQ:

add_car nomli procedure yozing.

  • Kirish: model VARCHAR, price INT, count INT
  • Vazifa: Yangi mashina qo'shish
  • Tekshiruvlar:
    • Model nomi bo'sh bo'lmasin
    • Model nomi takrorlanmasin (UNIQUE)
    • Narx musbat bo'lsin (> 0)
    • Count manfiy bo'lmasin (>= 0)
  • Model nomini avtomatik KATTA HARFga o'zgartiring
Misol:
CALL add_car('Lexus', 25000, 30); → LEXUS qo'shildi
CALL add_car('BMW', 15000, 10); → Xato: allaqachon mavjud
✅ Yechim:
CREATE OR REPLACE PROCEDURE add_car(
   p_model VARCHAR(255),
   p_price INT,
   p_count INT
)
LANGUAGE plpgsql
AS
$$
DECLARE
   model_exists BOOLEAN;
BEGIN
   -- Model nomini katta harfga
   p_model := UPPER(TRIM(p_model));
   
   -- Bo'sh nom tekshiruvi
   IF p_model = '' OR p_model IS NULL THEN
      RAISE NOTICE '❌ Model nomi bo''sh bo''lishi mumkin emas';
      ROLLBACK;
      RETURN;
   END IF;
   
   -- Takrorlanish tekshiruvi
   SELECT EXISTS(SELECT 1 FROM hw4.cars WHERE model = p_model)
   INTO model_exists;
   
   IF model_exists THEN
      RAISE NOTICE '❌ Bu model allaqachon mavjud: %', p_model;
      ROLLBACK;
      RETURN;
   END IF;
   
   -- Narx va count tekshiruvi
   IF p_price <= 0 THEN
      RAISE NOTICE '❌ Narx musbat bo''lishi kerak';
      ROLLBACK;
      RETURN;
   END IF;
   
   IF p_count < 0 THEN
      RAISE NOTICE '❌ Soni manfiy bo''lishi mumkin emas';
      ROLLBACK;
      RETURN;
   END IF;
   
   -- Qo'shish
   INSERT INTO hw4.cars(model, price, count)
   VALUES(p_model, p_price, p_count);
   
   COMMIT;
   RAISE NOTICE '✅ Yangi mashina qo''shildi: % (narx: %, soni: %)', 
      p_model, p_price, p_count;
END;
$$;

-- Ishlatish:
CALL add_car('Lexus', 25000, 30);   -- OK
CALL add_car('bmw', 15000, 10);     -- Xato
CALL add_car('', 10000, 5);         -- Xato

📝 Masala: Mashina audit log

▼
📋 TOPSHIRIQ:

Mashinalar jadvalidagi har qanday o'zgarishni qayd qiluvchi trigger yarating.

  • 1-qadam: hw4.car_audit_log jadvali yarating:
    • id SERIAL PRIMARY KEY
    • car_id INT
    • action VARCHAR(10) - INSERT/UPDATE/DELETE
    • old_price, new_price INT
    • old_count, new_count INT
    • changed_at TIMESTAMP DEFAULT NOW()
  • 2-qadam: log_car_changes() trigger function yarating
  • 3-qadam: AFTER INSERT OR UPDATE OR DELETE trigger yarating
Misol:
UPDATE hw4.cars SET price = 11000 WHERE id = 1;
car_audit_log ga avtomatik yoziladi:
action='UPDATE', old_price=10000, new_price=11000
✅ Yechim:
-- 1. Log jadvali yaratish
CREATE TABLE hw4.car_audit_log (
    id SERIAL PRIMARY KEY,
    car_id INT,
    action VARCHAR(10),
    old_model VARCHAR(255),
    new_model VARCHAR(255),
    old_price INT,
    new_price INT,
    old_count INT,
    new_count INT,
    changed_at TIMESTAMP DEFAULT NOW()
);

-- 2. Trigger function
CREATE OR REPLACE FUNCTION log_car_changes()
   RETURNS TRIGGER
   LANGUAGE plpgsql
AS
$$
BEGIN
   IF TG_OP = 'INSERT' THEN
      INSERT INTO hw4.car_audit_log(car_id, action, new_model, new_price, new_count)
      VALUES(NEW.id, 'INSERT', NEW.model, NEW.price, NEW.count);
      RETURN NEW;
      
   ELSIF TG_OP = 'UPDATE' THEN
      INSERT INTO hw4.car_audit_log(
         car_id, action, 
         old_model, new_model, 
         old_price, new_price, 
         old_count, new_count
      )
      VALUES(
         OLD.id, 'UPDATE', 
         OLD.model, NEW.model, 
         OLD.price, NEW.price, 
         OLD.count, NEW.count
      );
                            

                                2
                                

      INSERT INTO hw4.car_audit_log(car_id, action, old_model, old_price, old_count)
      VALUES(OLD.id, 'DELETE', OLD.model, OLD.price, OLD.count);
      RETURN OLD;
   END IF;
END;
$$;

-- 3. Triggerni jadvalga bog'lash
CREATE TRIGGER car_audit_trigger
   AFTER INSERT OR UPDATE OR DELETE
   ON hw4.cars
   FOR EACH ROW
   EXECUTE FUNCTION log_car_changes();

  • Tekshiruvlar:

CREATE OR REPLACE FUNCTION validate_car_data()

$$

   -- Narx tekshiruvi

   END IF;
   
   -- Count tekshiruvi
   IF NEW.count < 0 THEN
      RAISE EXCEPTION '❌ Soni manfiy bo''lishi mumkin emas! Kiritilgan: %', NEW.count;
   END IF;
   
   -- Model nomini katta harfga
                            

                                3
                                

   IF NEW.model = '' OR NEW.model IS NULL THEN
      RAISE EXCEPTION '❌ Model nomi bo''sh bo''lishi mumkin emas!';
   END IF;
   
   RETURN NEW;
END;
$$;

CREATE TRIGGER validate_car_trigger
   BEFORE INSERT OR UPDATE
   ON hw4.cars

📋 TOPSHIRIQ:

Bir martada narx 50% dan ko'p oshishini oldini oluvchi trigger yarating.

  • Faqat UPDATE da ishlaydi
  • Faqat narx oshganda tekshiradi (kamaytirish mumkin)
  • Foizni hisoblang: ((NEW.price - OLD.price) / OLD.price) * 100
  • 50% dan oshsa RAISE EXCEPTION
  • BEFORE UPDATE da ishlashi kerak

CREATE OR REPLACE FUNCTION check_price_increase()
   RETURNS TRIGGER
   LANGUAGE plpgsql
AS
$$
DECLARE
   increase_percent NUMERIC;
BEGIN
   -- Faqat UPDATE da va narx oshganda
   IF TG_OP = 'UPDATE' AND NEW.price > OLD.price THEN
      -- Foizni hisoblash
      increase_percent := ((NEW.price - OLD.price)::NUMERIC / OLD.price) * 100;

📝 Masala: Avtomatik chegirma

▼
📋 TOPSHIRIQ:

Agar mashina soni 100 dan oshsa, narxni avtomatik 5% kamaytiruvchi trigger yarating.

  • Shart: NEW.count > 100 VA OLD.count <= 100 (yoki INSERT)
  • Chegirma: NEW.price * 0.95
  • BEFORE INSERT OR UPDATE da ishlashi kerak
  • RAISE NOTICE bilan xabar bering
Misol:
UPDATE hw4.cars SET count = 150 WHERE id = 1;
BMW count: 50 → 150 (100 dan oshdi)
BMW price: 10000 → 9500 (avtomatik 5% chegirma)

INSERT INTO hw4.cars(model, price, count) VALUES('HONDA', 15000, 120);
HONDA price: 15000 → 14250 (avtomatik chegirma)
✅ Yechim:
CREATE OR REPLACE FUNCTION auto_discount()
   RETURNS TRIGGER
   LANGUAGE plpgsql
AS
$$
DECLARE
   discounted_price INT;
BEGIN
   -- Faqat count oshganda va 100 dan o'tganda
   IF NEW.count > 100 AND (OLD.count IS NULL OR OLD.count <= 100) THEN
      -- 5% chegirma
      discounted_price := ROUND(NEW.price * 0.95);
      
      RAISE NOTICE '🏷️ Avtomatik chegirma: % uchun narx % dan % ga tushdi', 
         NEW.model, NEW.price, discounted_price;
      
      NEW.price := discounted_price;
   END IF;
   
   RETURN NEW;
END;
$$;

CREATE TRIGGER auto_discount_trigger
   BEFORE INSERT OR UPDATE
   ON hw4.cars
   FOR EACH ROW
   EXECUTE FUNCTION auto_discount();
